# Modbus TCP reader
from pymodbus.client import ModbusTcpClient

def read_regs(host, addr, n):
    c = ModbusTcpClient(host)
    c.connect()
    r = c.read_holding_registers(
        addr, n, unit=1)
    c.close()
    return r.registers

data = read_regs(
    "192.168.1.5", 0, 10)
print(f"Regs: {data}")
Modbus TCP Reader
Read holding registers from any Modbus device

// ESP32 MQTT Publisher
#include <WiFi.h>
#include <PubSubClient.h>

WiFiClient espClient;
PubSubClient mqtt(espClient);

void publish(float temp) {
  String p = "{\"temp\":"
           + String(temp) + "}";
  mqtt.publish(
    "sensor/data", p.c_str());
}

PYTHON
# OPC-UA value reader
from opcua import Client

def read_opc(url, node_id):
    c = Client(url)
    try:
        c.connect()
        node = c.get_node(node_id)
        return node.get_value()
    finally:
        c.disconnect()

v = read_opc(
  "opc.tcp://localhost:4840",
  "ns=2;i=1")
print(f"Value: {v}")
OPC-UA Reader
Read node values from OPC-UA server
